# Patch Android manifest to add deep link support for OAuth callback
patch-android-manifest:
    #!/usr/bin/env bash
    MANIFEST="$(pwd)/code/target/dx/app/debug/android/app/app/src/main/AndroidManifest.xml"
    if [ -f "$MANIFEST" ]; then
        if ! grep -q "repforge" "$MANIFEST"; then
            # macOS compatible: use perl instead of sed for multi-line replace
            perl -i -p0e 's|</intent-filter>\s*</activity>|</intent-filter>
            <!-- Deep link for OAuth callback -->
            <intent-filter android:autoVerify="false">
                <action android:name="android.intent.action.VIEW" />
                <category android:name="android.intent.category.DEFAULT" />
                <category android:name="android.intent.category.BROWSABLE" />
                <data android:scheme="repforge" android:host="callback" />
            </intent-filter>
        </activity>|s' "$MANIFEST"
            echo "✓ Patched AndroidManifest.xml with deep link support"
        else
            echo "✓ AndroidManifest.xml already has deep link support"
        fi
    else
        echo "⚠ AndroidManifest.xml not found - run android build first"
    fi

# Build and run Android app (requires backend via Pulumi)
# Use `just android` for dev stack (7xxx) or `just android 9` for e2e stack (9xxx)
android stack="7":
    #!/usr/bin/env bash
    set -e
    ROOT="$(pwd)"
    STACK="{{stack}}"
    if [ "$STACK" = "9" ]; then
        KC_PORT=9051; FS_PORT=9081
    else
        KC_PORT=7051; FS_PORT=7081
    fi
    # Start emulator if not running
    if ! adb devices | grep -q "emulator"; then
        echo "Starting Android emulator..."
        ~/Library/Android/sdk/emulator/emulator @Test_phone &
        adb wait-for-device
        echo "Waiting for emulator to boot..."
        while [ "$(adb shell getprop sys.boot_completed 2>/dev/null)" != "1" ]; do sleep 1; done
        echo "Emulator ready"
    fi
    # Set up adb reverse tunnels so emulator can reach host services via localhost
    echo "Setting up adb reverse tunnels..."
    adb reverse tcp:$KC_PORT tcp:$KC_PORT  # Keycloak
    adb reverse tcp:$FS_PORT tcp:$FS_PORT  # Fullstack server
    echo "✓ Tunnels: localhost:$KC_PORT (Keycloak), localhost:$FS_PORT (Fullstack)"
    # Export env vars so the native app picks up the right stack
    export REPFORGE_KEYCLOAK_URL="http://localhost:$KC_PORT"
    export REPFORGE_FRONTEND_BASE_URL="http://localhost:$FS_PORT"
    export REPFORGE_REDIRECT_URI="http://localhost:$FS_PORT/callback"
    cd "$ROOT/code/frontend/app" && dx build --platform android --features mobile
    cd "$ROOT"
    just patch-android-manifest
    APK="$ROOT/code/target/dx/app/debug/android/app/app/build/outputs/apk/debug/app-debug.apk"
    adb uninstall com.repforge.app || true
    adb install "$APK"
    adb shell am start -n com.repforge.app/dev.dioxus.main.MainActivity

# Dev stack (Pulumi/Docker)
#   just dev                         - start shared deps + all backend service containers
#   just dev deps                    - start shared deps only
#   just dev user-api                - start shared deps + only user-api
#   just dev workout-api             - start shared deps + only workout-api
#   just dev exercise-api            - start shared deps + only exercise-api
#   just dev down                    - destroy the dev stack
#   just dev down user-api           - disable one API container, keep shared deps up
dev *args:
    #!/usr/bin/env bash
    set -euo pipefail
    cd "{{justfile_directory()}}"
    set -- {{args}}

    stack="cqrs"

    usage() {
        cat <<'EOF'
    Usage:
      just dev                         Start dev infra + all backend APIs
      just dev deps                    Start shared deps only
      just dev user-api                Start shared deps + only user-api
      just dev workout-api             Start shared deps + only workout-api
      just dev exercise-api            Start shared deps + only exercise-api
      just dev down                    Destroy the dev stack
      just dev down user-api           Disable one API container and keep shared deps up

    Dev ports:
      Keycloak      http://localhost:7051
      User API      http://localhost:7061
      Workout API   http://localhost:7062
      Exercise API  http://localhost:7063
      Fullstack     http://localhost:7081 (run with: just local fullstack)
      Elasticsearch http://localhost:7200
      Kafka UI      http://localhost:7052
      Mailpit       http://localhost:7057
      Prometheus    http://localhost:7090
    EOF
    }

    pulumi_cmd() {
        (cd infra && PULUMI_CONFIG_PASSPHRASE='' pulumi "$@")
    }

    ensure_stack() {
        (cd infra && npm install --silent)
        pulumi_cmd stack select "$stack" 2>/dev/null || pulumi_cmd stack init "$stack" --secrets-provider passphrase
    }

    select_stack() {
        pulumi_cmd stack select "$stack" >/dev/null
    }

    set_config() {
        pulumi_cmd config set "$1" "$2" -s "$stack" >/dev/null
    }

    set_services() {
        set_config enableFullstack false
        set_config enableUserApi "$1"
        set_config enableWorkoutApi "$2"
        set_config enableExerciseApi "$3"
    }

    normalize_service() {
        case "$1" in
            user|users|user-api) echo "user-api" ;;
            workout|workouts|workout-api) echo "workout-api" ;;
            exercise|exercises|exercise-api) echo "exercise-api" ;;
            deps|infra|dependencies) echo "deps" ;;
            all|"") echo "all" ;;
            *) echo "" ;;
        esac
    }

    enable_only_service() {
        case "$1" in
            user-api) set_services true false false ;;
            workout-api) set_services false true false ;;
            exercise-api) set_services false false true ;;
            deps) set_services false false false ;;
            all) set_services true true true ;;
            *) echo "Unknown dev service: $1" >&2; usage; exit 2 ;;
        esac
    }

    disable_service() {
        case "$1" in
            user-api) set_config enableUserApi false ;;
            workout-api) set_config enableWorkoutApi false ;;
            exercise-api) set_config enableExerciseApi false ;;
            deps|all) ;;
            *) echo "Unknown dev service: $1" >&2; usage; exit 2 ;;
        esac
    }

    wait_http() {
        local name="$1"
        local url="$2"
        local limit="${3:-60}"
        echo "==> Waiting for ${name}..."
        for i in $(seq 1 "$limit"); do
            if curl -sf "$url" >/dev/null 2>&1; then
                echo "    ${name} ready after ${i}s"
                return 0
            fi
            sleep 1
        done
        echo "    ${name} did not become ready within ${limit}s" >&2
        return 1
    }

    wait_dev_stack() {
        local target="$1"
        wait_http "Keycloak" "http://localhost:7051/realms/local/.well-known/openid-configuration" 120
        wait_http "Elasticsearch" "http://localhost:7200/_cluster/health" 120
        case "$target" in
            all)
                wait_http "user-api" "http://localhost:7061/ready" 60
                wait_http "workout-api" "http://localhost:7062/ready" 60
                wait_http "exercise-api" "http://localhost:7063/ready" 60
                ;;
            user-api) wait_http "user-api" "http://localhost:7061/ready" 60 ;;
            workout-api) wait_http "workout-api" "http://localhost:7062/ready" 60 ;;
            exercise-api) wait_http "exercise-api" "http://localhost:7063/ready" 60 ;;
            deps) ;;
        esac
    }

    print_urls() {
        cat <<'EOF'
    Dev stack URLs:
      Keycloak      http://localhost:7051
      User API      http://localhost:7061
      Workout API   http://localhost:7062
      Exercise API  http://localhost:7063
      Elasticsearch http://localhost:7200
      Kafka UI      http://localhost:7052
      Mailpit       http://localhost:7057
      Prometheus    http://localhost:7090
    EOF
    }

    # Route: just dev → start all; just dev <service> → start service; just dev down [service] → stop
    action="${1:-}"
    if [ "$action" = "down" ]; then
        target="$(normalize_service "${2:-all}")"
        if [ "$target" = "all" ]; then
            select_stack
            pulumi_cmd destroy -s "$stack" --yes --skip-preview
            docker rmi repforge/user-api:latest repforge/workout-api:latest repforge/exercise-api:latest 2>/dev/null || true
        elif [ -n "$target" ]; then
            select_stack
            disable_service "$target"
            pulumi_cmd up -s "$stack" --yes --skip-preview
            echo "Disabled ${target}; shared dev infra remains up."
        else
            echo "Unknown dev target: ${2:-}" >&2
            usage
            exit 2
        fi
    elif [ "$action" = "help" ] || [ "$action" = "-h" ] || [ "$action" = "--help" ]; then
        usage
    else
        # No action or a service name: start the stack
        target="$(normalize_service "${action:-all}")"
        if [ -z "$target" ]; then
            echo "Unknown dev target: ${action:-}" >&2
            usage
            exit 2
        fi
        ensure_stack
        enable_only_service "$target"
        pulumi_cmd up -s "$stack" --yes --skip-preview
        wait_dev_stack "$target"
        print_urls
    fi

# Audit visual manifest against linux/darwin baseline directories
test-visual-audit:
    #!/usr/bin/env bash
    set -euo pipefail
    ROOT="$(pwd)"
    MANIFEST="$ROOT/code/tests/e2e/visual-manifest.toml"
    LINUX_DIR="$ROOT/code/tests/e2e/baselines/linux"
    DARWIN_DIR="$ROOT/code/tests/e2e/baselines/darwin"
    TMP_DIR="$(mktemp -d)"
    trap 'rm -rf "$TMP_DIR"' EXIT

    if [ ! -f "$MANIFEST" ]; then
      echo "visual manifest not found: $MANIFEST"
      exit 1
    fi

    rg -o '"[A-Za-z0-9_-]+"' "$MANIFEST" | tr -d '"' | sort -u > "$TMP_DIR/manifest.txt"
    if [ -d "$LINUX_DIR" ]; then
      find "$LINUX_DIR" -maxdepth 1 -type f -name '*.png' -exec basename {} .png \; | sort -u > "$TMP_DIR/linux.txt"
    else
      touch "$TMP_DIR/linux.txt"
    fi

    if [ -d "$DARWIN_DIR" ]; then
      find "$DARWIN_DIR" -maxdepth 1 -type f -name '*.png' -exec basename {} .png \; | sort -u > "$TMP_DIR/darwin.txt"
    else
      touch "$TMP_DIR/darwin.txt"
    fi

    cat "$TMP_DIR/linux.txt" "$TMP_DIR/darwin.txt" | sort -u > "$TMP_DIR/available.txt"

    comm -23 "$TMP_DIR/manifest.txt" "$TMP_DIR/available.txt" > "$TMP_DIR/missing-any.txt" || true
    comm -23 "$TMP_DIR/manifest.txt" "$TMP_DIR/linux.txt" > "$TMP_DIR/missing-linux.txt" || true
    comm -13 "$TMP_DIR/manifest.txt" "$TMP_DIR/linux.txt" > "$TMP_DIR/orphan-linux.txt" || true
    comm -13 "$TMP_DIR/manifest.txt" "$TMP_DIR/darwin.txt" > "$TMP_DIR/orphan-darwin.txt" || true

    HAS_ISSUES=0
    if [ -s "$TMP_DIR/missing-any.txt" ]; then
      HAS_ISSUES=1
      echo "Missing baselines (not found in linux or darwin):"
      cat "$TMP_DIR/missing-any.txt"
    fi
    if [ -s "$TMP_DIR/missing-linux.txt" ]; then
      if [ "${VISUAL_STRICT_LINUX_CANONICAL:-0}" = "1" ]; then
        HAS_ISSUES=1
        echo "Missing linux canonical baselines:"
        cat "$TMP_DIR/missing-linux.txt"
      else
        echo "Warning: missing linux canonical baselines (using darwin fallback):"
        cat "$TMP_DIR/missing-linux.txt"
      fi
    fi
    if [ -s "$TMP_DIR/orphan-linux.txt" ]; then
      HAS_ISSUES=1
      echo "Orphan linux baselines (not in manifest):"
      cat "$TMP_DIR/orphan-linux.txt"
    fi
    if [ -s "$TMP_DIR/orphan-darwin.txt" ]; then
      HAS_ISSUES=1
      echo "Orphan darwin baselines (not in manifest):"
      cat "$TMP_DIR/orphan-darwin.txt"
    fi

    if [ "$HAS_ISSUES" -ne 0 ]; then
      exit 1
    fi

    echo "Visual audit passed: manifest and baseline files are aligned."

# Validate required canonical database environment variables for local/e2e runs
validate-env:
    #!/usr/bin/env bash
    set -euo pipefail
    REQUIRED_VARS=(
      APP_DATABASE_URL
      USER_API_DATABASE_URL
      WORKOUT_API_DATABASE_URL
      EXERCISE_API_DATABASE_URL
    )

    MISSING=()
    for VAR in "${REQUIRED_VARS[@]}"; do
      if [ -z "${!VAR:-}" ]; then
        MISSING+=("$VAR")
      fi
    done

    if [ "${#MISSING[@]}" -ne 0 ]; then
      echo "Missing required canonical env vars:"
      printf ' - %s\n' "${MISSING[@]}"
      exit 1
    fi

    echo "Environment validation passed."

# Check for CSS vars that have semantic token equivalents
check-tokens:
    #!/usr/bin/env bash
    echo "Checking for CSS vars that should use semantic tokens..."
    # Only flag specific patterns that have semantic token equivalents:
    # - var(--color-primary-100) -> colors::semantic::PRIMARY_LIGHT
    # - var(--color-primary-700) -> colors::semantic::PRIMARY_DARK
    # - var(--color-on-primary)  -> colors::semantic::ON_PRIMARY
    # - var(--color-bg-overlay)  -> colors::semantic::BG_OVERLAY
    VIOLATIONS=$(grep -rE --include="*.rs" "var\(--color-(primary-100|primary-700|on-primary|bg-overlay|primary-light|primary-dark)\)" code/frontend/ | grep -v "/tokens/" | grep -v "// allow-raw-var" || true)
    if [ -n "$VIOLATIONS" ]; then
        echo "❌ Found CSS vars that should use semantic tokens:"
        echo "$VIOLATIONS"
        echo ""
        echo "Use semantic tokens instead:"
        echo "  var(--color-primary-100)  -> colors::semantic::PRIMARY_LIGHT"
        echo "  var(--color-primary-700)  -> colors::semantic::PRIMARY_DARK"
        echo "  var(--color-on-primary)   -> colors::semantic::ON_PRIMARY"
        echo "  var(--color-bg-overlay)   -> colors::semantic::BG_OVERLAY"
        echo "  var(--color-primary-light)-> colors::semantic::PRIMARY_LIGHT"
        echo "  var(--color-primary-dark) -> colors::semantic::PRIMARY_DARK"
        echo ""
        echo "Or add '// allow-raw-var' comment if intentional."
        exit 1
    else
        echo "✓ No semantic token violations found"
    fi

# Internal: Start e2e infrastructure (9xxx ports)
_e2e-infra-up:
    #!/usr/bin/env bash
    set -euo pipefail
    cd "{{justfile_directory()}}"
    echo "==> Starting e2e infrastructure on 9xxx ports..."
    (cd infra && npm install --silent)
    # Initialize stack if it doesn't exist
    (cd infra && PULUMI_CONFIG_PASSPHRASE='' pulumi stack select e2e 2>/dev/null || PULUMI_CONFIG_PASSPHRASE='' pulumi stack init e2e --secrets-provider passphrase)
    # Docker Desktop can deadlock under parallel BuildKit image builds on the
    # e2e stack. Keep stack startup deterministic; test concurrency still comes
    # from nextest once the infrastructure is healthy.
    (cd infra && PULUMI_CONFIG_PASSPHRASE='' pulumi up -s e2e --yes --skip-preview --parallel 1) || {
        echo "Failed to start e2e stack"
        exit 1
    }

# Internal: Stop e2e infrastructure
_e2e-infra-down:
    #!/usr/bin/env bash
    cd "{{justfile_directory()}}"
    echo "==> Stopping e2e infrastructure..."
    # All services (including fullstack) are managed by Pulumi
    (cd infra && PULUMI_CONFIG_PASSPHRASE='' pulumi destroy -s e2e --yes --skip-preview 2>/dev/null) || true
    # Optional: remove stack state entirely (set E2E_STACK_RM=1 when needed)
    if [ "${E2E_STACK_RM:-0}" = "1" ]; then
        (cd infra && PULUMI_CONFIG_PASSPHRASE='' pulumi stack rm e2e --yes 2>/dev/null) || true
    fi
    # Remove cached Docker images so they're rebuilt from scratch
    docker rmi repforge/user-api:latest repforge/workout-api:latest repforge/exercise-api:latest repforge/fullstack:latest 2>/dev/null || true

# Internal: Wait for services to be healthy (all deployed via Pulumi)

    #!/usr/bin/env bash
    set -euo pipefail
    echo "==> Waiting for Keycloak (9051)..."
    for i in $(seq 1 120); do
        if curl -sf http://localhost:9051/realms/local/.well-known/openid-configuration > /dev/null 2>&1; then
            echo "    Keycloak ready after $i seconds"
            break
        fi
        if [ $i -eq 120 ]; then
